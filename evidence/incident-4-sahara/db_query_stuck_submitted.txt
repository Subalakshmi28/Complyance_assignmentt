-- Database inspection for Sahara Medical import
SELECT status, count(*), count(NULLIF(asp_message_id, '')) AS with_asp_ref
FROM invoices
WHERE seller_id = 5 AND invoice_number LIKE 'SMS-1116-%'
GROUP BY status;

-- Result:
-- status    | count | with_asp_ref
-- Accepted  | 8     | 8
-- Submitted | 32    | 0
